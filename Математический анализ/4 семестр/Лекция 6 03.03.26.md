    \]
    Длина дуги, при $\gamma: (x(t), y(t), z(t))$, $t \in [a; b]$.

    \begin{tikzpicture}[baseline=(current bounding box.center)]

    \[
    \oint\limits_{\partial D} P dx + Q dy = \iint\limits_{D} \left( \frac{\partial Q}{\partial x} - \frac{\partial P}{\partial y} \right) dx dy
    \]
    \begin{enumerate}
        \item $P=0, \quad Q=x \quad \oint\limits_{\partial D} x dy = \iint\limits_{D} 1 \, dx = S(D)$ \\
        Т.е. $S(D) = \oint\limits_{\partial D} x dy$
        
        \item $P=y, \quad Q=0 \quad \oint\limits_{\partial D} y dx = - \iint\limits_{D} 1 \, dx dy = -S(D)$ \\
        $S(D) = - \oint\limits_{\partial D} y dx$
        
        \item $P=y, \quad Q=-x$ \\
        $\oint\limits_{\partial D} y dx - x dy = - \iint\limits_{D} 2 dx dy$ \\
        Т.е. $S(D) = \frac{1}{2} \oint\limits_{\partial D} x dy - y dx$
    \end{enumerate}
\end{enumerate}

\section*{Вспомним формулу Грина:}
\[
\oint\limits_{\partial D} P dx + Q dy = \iint\limits_{D} \left( \frac{\partial Q}{\partial x} - \frac{\partial P}{\partial y} \right) dx dy
\]

\textbf{Определение.} $\omega(x, y) = P(x, y) dx + Q(x, y) dy$ - дифференциальная форма 1-ого порядка.

\textbf{Определение.} $d\omega(x, y) = \left( \frac{\partial Q}{\partial x} - \frac{\partial P}{\partial y} \right) dx dy$ - дифференциал дифференциальной формы 1-ого порядка.

\begin{align*}
d\omega(x, y) &= \frac{\partial (P dx + Q dy)}{\partial x} dx + \frac{\partial (P dx + Q dy)}{\partial y} dy = \\
&= (\text{считают, что } dx dx = 0; \ dy dy = 0; \ dx dy = -dy dx) \\
&= \frac{\partial P}{\partial x} dx dx + \frac{\partial Q}{\partial x} dx dy + \frac{\partial P}{\partial y} dy dx + \frac{\partial Q}{\partial y} dy dy = \\
&= \left( \frac{\partial Q}{\partial x} - \frac{\partial P}{\partial y} \right) dx dy
\end{align*}

\[
\oint\limits_{\partial D} \omega = \iint\limits_{D} d\omega \quad \text{-- другая форма формулы Грина}
\]

$\vec{r}(u, v); (u, v) \in K$.\\
$\vec{r}(u_0, v_0) = \vec{r}_0$.\\
$\vec{r}(u(t), v(t)) : \gamma$.\\
$(u_0, v_0) = (u(t_0), v(t_0))$.\\
$(\vec{r}'_u = \vec{r}_u)$.

$\vec{r}(t) = (\varphi_1(u, v); \varphi_2(u, v); \varphi_3(u, v))$.

\begin{align*}
\frac{d\vec{r}(t)}{dt} &= \left( \frac{\partial \varphi_1}{\partial u} \cdot \frac{du}{dt} + \frac{\partial \varphi_1}{\partial v} \cdot \frac{dv}{dt}; \frac{\partial \varphi_2}{\partial u} \cdot \frac{du}{dt} + \right. \\
& \left. + \frac{\partial \varphi_2}{\partial v} \frac{dv}{dt}; \frac{\partial \varphi_3}{\partial u} \cdot \frac{du}{dt} + \frac{\partial \varphi_3}{\partial v} \cdot \frac{dv}{dt} \right) = \\
&= \left( \frac{\partial \vec{r}(u(t), v(t))}{\partial u}; \frac{\partial \vec{r}(u(t), v(t))}{\partial v} \right) \cdot \begin{pmatrix} u'(t) \\ v'(t) \end{pmatrix} = \\
&= \vec{r}'_u(u(t), v(t)) \cdot u'(t) + \vec{r}'_v(u(t), v(t)) \cdot v'(t)
\end{align*}

$\frac{d\vec{r}(t_0)}{dt} = \vec{r}_u(u(t_0), v(t_0)) \cdot u'(t_0) + \vec{r}_v(u(t_0), v(t_0)) \cdot v'(t_0)$.

\textbf{Определение.} Касательной плоскостью к поверхности $S$ будет называть плоскость, в которой лежат все касательные векторы к кривым $S$ через заданную точку.
